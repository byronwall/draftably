diagram "Checkout service architecture" version=1
subtitle "Orders are written synchronously, then updated by a worker"

group clients "Clients" color=blue
node web "Web app" in=clients
node mobile "Mobile app" in=clients

group services "Services" color=mint
node gateway "API gateway" in=services
node auth "Auth service" in=services
node orders "Order service" in=services
node worker "Worker" in=services

group data "Data" color=violet
node orders_db "Orders database" in=data
node queue "Queue" in=data shape=pill

legend blue "Clients"
legend mint "Services"
legend violet "Data layer"
legend line solid "Request path"
legend line dashed "Asynchronous messaging"

edge web_api web -> gateway "HTTPS"
edge mobile_api mobile -> gateway "HTTPS"
edge validate gateway -> auth "validate token"
edge route gateway -> orders "send request"
edge rw orders -> orders_db "read / write"
edge publish orders -> queue "publish" kind=async
edge consume queue -> worker "consume" kind=async
edge update worker -> orders_db "update"

layout root flow gap=48 fit=slide
