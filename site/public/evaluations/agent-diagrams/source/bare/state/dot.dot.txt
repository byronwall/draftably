digraph PublishingStateMachine {
    label="Publishing state machine";
    labelloc="t";
    fontsize=20;
    rankdir=LR;
    node [shape=box, style=rounded];

    start [shape=point, width=0.15];

    draft [label="Draft"];
    in_review [label="In Review"];
    changes_requested [label="Changes Requested"];
    approved [label="Approved"];
    published [label="Published"];
    archived [label="Archived"];

    start -> draft;
    draft -> in_review [label="submit"];
    in_review -> changes_requested [label="request edits"];
    changes_requested -> draft [label="revise"];
    in_review -> approved [label="approve"];
    approved -> published [label="publish"];
    approved -> draft [label="withdraw"];
    published -> archived [label="archive"];
    archived -> draft [label="restore"];
}
